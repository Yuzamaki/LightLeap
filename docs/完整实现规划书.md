```Python
def classify_food(food_item):
    """
    基于中国居民膳食指南（2022）的简化着色逻辑
    核心：不关注精确卡路里，关注营养密度
    """
    # 绿灯：高营养密度，随意吃
    if food_item.category in ['蔬菜', '水果', '优质蛋白', '全谷物']:
        return 'green'
    
    # 黄灯：中等营养密度，适量
    if food_item.category in ['精制主食', '瘦肉', '豆制品', '乳制品']:
        return 'yellow'

    if food_item.category in ['油炸', '甜点', '含糖饮料', '高脂零食']:
        return 'red'
    
    # 混合食物（如火锅）→ 按主要成分比例决策
    # 由 AI 模型按成分拆分后分类评级
    return ai_mixed_classify(food_item)
```

### **2\.4 体重与趋势**



#### **实现方案**



- 每周建议称重 1\-2 次（不在同一天）

- 输入方式：手动输入或连接蓝牙体脂秤

- 图表实现：`fl_chart` 库的 `LineChart`

\- **核心算法：移动平均平滑**



```Python
def smooth_weight(raw_weights: list[tuple[date, float]]) -> list[tuple[date, float]]:
    """
    7 日移动平均 + 异常值过滤
    """
    sorted_weights = sorted(raw_weights, key=lambda x: x[0])
    daily = defaultdict(list)
    for d, w in sorted_weights:
        daily[d].append(w)
    
    dates = sorted(daily.keys())
    avg_weights = [sum(daily[d])/len(daily[d]) for d in dates]
    
    # 中位数移动平均（窗口=3）
    smoothed = []
    for i in range(len(dates)):
        left = max(0, i-3)
        right = min(len(dates), i+4)
        window = avg_weights[left:right]
        smoothed.append(statistics.median(window))
    
    return list(zip(dates, smoothed))
```

```Python
    }
    
    # 1. 分析完成率
    completion_rate = analyze_completion_rate(user.last_week_plan, user.last_week_actual)
    
    # 2. 调整难度
    if completion_rate > 0.8:
        difficulty = "maintain"        # 维持或略增
    elif completion_rate > 0.5:
        difficulty = "slightly_easier" # 稍微降低
    else:
        difficulty = "much_easier"     # 大幅降低，先建立信心
    
    # 3. 考虑外部因素
    weather = get_weekly_forecast(user.city_level)  # 仅城市级
    if weather.has_rain_days:
        for rainy_day in weather.rainy_days:
            week_plan[rainy_day].exercise = "indoor"  # 室内运动提案
    
    # 4. 考虑生理期（女性用户，且授权记录）
    if user.gender == 'female' and user.has_mens_tracking:
        cycle = predict_cycle(user.mens_history)
        if cycle.is_menstrual_phase:
            week_plan["运动建议"] = "以散步/瑜伽为主，避免高强度"
    
    # 5. 考虑学业压力（简间接：通过运动频率下降 + 夜间活跃度判断）
    if detected_academic_pressure(user):
        week_plan["建议"] = "这个月貌似很忙？运动可以短一点，15 分钟HIIT也很棒"
    
    # 6. 生成具体内容
    for day in week_days:
        week_plan[day] = generate_day_plan(user, difficulty, rest_days=["wednesday"])
    
    # 7. 输出自然语言建议 + 结构化数据
    return {
        "structured": week_plan,
        "nl_summary": generate_nl_summary(week_plan, user.preference)  # AI 润色
    }
```

### **6\.3 LLM 对话工程**



```Python
# 调用架构

import openai
from typing import Optional

class AIHealthCoach:
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.user_context = self._load_context()
    
    def _load_context(self) -> dict:
        """加载用户上下文（摘要级，不包含原始数据）"""
        return {
            "age_group": fetch_user_age_group(self.user_id),
            "weekly_veggie_rate": get_weekly_metric(self.user_id, "veggie_rate"),
            "weekly_exercise_mins": get_weekly_metric(self.user_id, "exercise_minutes"),
            "consecutive_days": get_streak(self.user_id),
            "mood_trend": get_recent_mood_trend(self.user_id),
        }
    
    async def chat(self, message: str) -> str:
        # 1. 安全检查（输入）
        if self._input_unsafe(message):
            return self._safe_redirect(message)
        
        # 2. 构建 prompt
        messages = [
            {"role": "system", "content": build_system_prompt(self.user_context)},
            *self._recent_history(limit=10),   # 最近 10 轮
            {"role": "user", "content": message},
        ]
        
        # 3. 调用 LLM
        response = await openai.ChatCompletion.acreate(
            model="gpt-4o-mini",
            messages=messages,
            max_tokens=500,
            temperature=0.7,
        )
        reply = response.choices[0].message.content
        
        # 4. 安全检查（输出）
        if self._output_unsafe(reply):
            reply = self._fallback_reply()
            self._alert_safety_team(self.user_id, reply)
        
        # 5. 保存历史
        await self._save_history(message, reply)
        
        return reply
```
